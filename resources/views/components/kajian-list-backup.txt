<?php

use Livewire\Component;
use Livewire\WithPagination;
use Livewire\Attributes\Url;

new class extends Component {
    use WithPagination;

    #[Url]
    public $search = '';

    #[Url]
    public $category = '';

    public function updatingSearch()
    {
        $this->resetPage();
    }

    public function updatingCategory()
    {
        $this->resetPage();
    }

    public function setCategory($slug)
    {
        $this->category = $slug;
        $this->resetPage();
    }

    public function resetSearch()
    {
        $this->search = '';
        $this->resetPage();
    }

    public function resetFilters()
    {
        $this->reset(['search', 'category']);
        $this->resetPage();
    }

    public function with(): array
    {
        $query = \App\Models\KajianDetail::with(['kajian.kajianCategory', 'ustadz', 'location'])
            ->whereHas('kajian', function ($q) {
                $q->whereNull('deleted_at');
            });

        if ($this->category) {
            $query->whereHas('kajian.kajianCategory', function ($q) {
                $q->where('slug', $this->category);
            });
        }

        if ($this->search) {
            $query->where(function ($q) {
                $q->whereHas('kajian', function ($q2) {
                    $q2->where('title', 'like', '%' . $this->search . '%');
                })->orWhere('sub_title', 'like', '%' . $this->search . '%')
                  ->orWhereHas('ustadz', function ($q3) {
                    $q3->where('name', 'like', '%' . $this->search . '%');
                  });
            });
        }

        $kajianDetails = $query->orderBy('date', 'asc')->paginate(9);
        $categories = \App\Models\KajianCategory::orderBy('name')->get();

        return [
            'kajianDetails' => $kajianDetails,
            'categories' => $categories,
        ];
    }
}; ?>

<main id="kajian-all" class="bg-slate-50 py-16">

            <p class="text-slate-500 text-sm mt-3 max-w-lg mx-auto">Cari kajian berdasarkan judul, nama ustadz, atau filter berdasarkan kategori.</p>

                    <input wire:model.live.debounce.400ms="search" type="search"
                        placeholder="Cari kajian atau ustadz..."
                        class="w-full pl-11 pr-4 py-3 rounded-xl border border-slate-200 focus:border-yellow-400 focus:ring focus:ring-yellow-400/20 transition-all text-sm">
                </div>

                @if($search || $category)
                    <button wire:click="resetFilters"

                        <iconify-icon icon="mdi:close"></iconify-icon>

                    Semua Kajian

                        {{ $cat->name }}

            <p class="text-sm text-slate-500">
                Menampilkan <span class="font-bold text-slate-800">{{ $kajianDetails->total() }}</span> kajian
                @if($search) untuk "<span class="font-bold text-yellow-600">{{ $search }}</span>" @endif
            </p>

                Memuat...

                <p class="text-slate-500 text-sm max-w-sm mx-auto">Tidak ditemukan jadwal kajian yang cocok dengan pencarian atau filter Anda.</p>
                <button type="button" wire:click="resetFilters"

                    <iconify-icon icon="mdi:refresh"></iconify-icon>

            <div class="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">

                    @php

                            'fiqih', 'fiqih-muamalah' => ['bg' => 'bg-emerald-500', 'text' => 'text-emerald-600', 'light' => 'bg-emerald-50 text-emerald-700 border-emerald-100'],
                            'sirah-nabawiyah' => ['bg' => 'bg-amber-500', 'text' => 'text-amber-600', 'light' => 'bg-amber-50 text-amber-700 border-amber-100'],
                            default => ['bg' => 'bg-yellow-500', 'text' => 'text-yellow-600', 'light' => 'bg-yellow-50 text-yellow-700 border-yellow-100']
                        };
                        $posterUrl = $detail->poster ? asset('storage/' . $detail->poster) : null;
                        $isPast = \Carbon\Carbon::parse($detail->date)->isPast();
                    @endphp

                    <a href="{{ route('kajian.show', $detail->id) }}" wire:navigate

                        {{-- Card Thumbnail / Header --}}

                                <img src="{{ $posterUrl }}" alt="{{ $detail->sub_title }}"

                            @else

                                    <div class="flex items-center justify-center h-full">
                                        <iconify-icon icon="mdi:book-open-variant" class="text-white/20 text-7xl"></iconify-icon>
                                    </div>
                                </div>
                            @endif

                            {{-- Badges --}}
                            <div class="absolute top-3 left-3 right-3 flex items-center justify-between gap-2">
                                <span class="bg-white/90 backdrop-blur-sm {{ $accentColor['text'] }} text-[10px] font-bold px-2.5 py-1 rounded-lg shadow-sm uppercase leading-none">
                                    {{ $detail->kajian->kajianCategory->name ?? 'Kajian' }}
                                </span>
                                @if($detail->location->type === 'Online')
                                    <span class="bg-red-500 text-white text-[10px] font-bold px-2.5 py-1 rounded-lg shadow-sm flex items-center gap-1 leading-none">
                                        <span class="w-1.5 h-1.5 bg-white rounded-full animate-ping"></span> LIVE

                                @elseif($isPast)

                            </div>

                            {{-- Ustadz Avatar + Name bottom of image --}}
                            <div class="absolute bottom-3 left-3 flex items-center gap-2">

                                    {{ substr($detail->ustadz->name ?? 'U', 0, 1) }}

        @if($kajianDetails->hasPages())

                {{ $kajianDetails->links() }}
            </div>
        @endif

    </div>
